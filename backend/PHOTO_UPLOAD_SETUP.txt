# PHOTO UPLOAD SETUP INSTRUCTIONS

## 1. Upload backend files to VPS

scp app/routes/upload.py root@148.135.138.145:/var/www/avenza/backend/app/routes/
scp app/main.py root@148.135.138.145:/var/www/avenza/backend/app/


## 2. Create uploads directory on VPS

ssh root@148.135.138.145

mkdir -p /var/www/avenza/frontend/uploads
chmod 755 /var/www/avenza/frontend/uploads
chown www-data:www-data /var/www/avenza/frontend/uploads


## 3. Update Nginx configuration

nano /etc/nginx/sites-available/avenza.live

# Add this location block inside the server block (after frontend root location):

    # Uploads folder
    location /uploads/ {
        alias /var/www/avenza/frontend/uploads/;
        expires 30d;
        add_header Cache-Control "public, immutable";
    }


## 4. Test Nginx configuration

nginx -t


## 5. Reload Nginx

systemctl reload nginx


## 6. Restart backend

sudo systemctl restart avenza


## 7. Test upload endpoint

curl -X POST https://avenza.live/api/upload/profile-photo \
  -H "Authorization: Bearer YOUR_TOKEN" \
  -F "file=@/path/to/test-image.jpg"


## COMPLETE NGINX CONFIG EXAMPLE:

server {
    listen 80;
    server_name avenza.live www.avenza.live;
    return 301 https://$server_name$request_uri;
}

server {
    listen 443 ssl http2;
    server_name avenza.live www.avenza.live;

    ssl_certificate /etc/letsencrypt/live/avenza.live/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/avenza.live/privkey.pem;

    # Frontend - Serve from dist folder
    location / {
        root /var/www/avenza/frontend;
        try_files $uri $uri/ /index.html;
        
        # Security headers
        add_header X-Frame-Options "SAMEORIGIN" always;
        add_header X-Content-Type-Options "nosniff" always;
        add_header X-XSS-Protection "1; mode=block" always;
    }

    # Uploads folder
    location /uploads/ {
        alias /var/www/avenza/frontend/uploads/;
        expires 30d;
        add_header Cache-Control "public, immutable";
    }

    # Backend API
    location /api/ {
        proxy_pass http://localhost:8028/api/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}
